cmake_minimum_required(VERSION 3.16)
project(h102_iio_examples LANGUAGES CXX)
find_path(IIO_INCLUDE_DIR NAMES iio.h REQUIRED)
find_library(IIO_LIBRARY NAMES iio libiio.so.0 REQUIRED)
add_executable(h102_iio h102_iio.cpp)
target_compile_features(h102_iio PRIVATE cxx_std_17)
target_include_directories(h102_iio PRIVATE ${IIO_INCLUDE_DIR})
target_link_libraries(h102_iio PRIVATE ${IIO_LIBRARY})
target_compile_options(h102_iio PRIVATE -Wall -Wextra -Wpedantic)
